# 12. ER図

AI Provider Abstraction Platform（APAP）設計書 第12章（PlantUML）

ID列はULID（26文字）。`(ES)` 印のAggregateはEvent Store併用（本テーブルは最新状態のスナップショット/Read Model）。

---

```plantuml
@startuml er_apap
hide circle
skinparam linetype ortho

entity PROVIDER <<ES>> {
  * provider_id : CHAR(26) <<PK>>
  --
  * name : VARCHAR(100) <<UQ>>
  * adapter_plugin_id : CHAR(26) <<FK>>
  * spi_version : VARCHAR(20)
  * auth_type : VARCHAR(40)
  * priority : SMALLINT
  * status : VARCHAR(20)
  * rate_limit_rpm : INT
  * rate_limit_tpm : INT
  * rate_limit_concurrent : INT
  tags : JSON
  * created_at / updated_at : TIMESTAMP
}

entity PROVIDER_ENDPOINT {
  * endpoint_id : CHAR(26) <<PK>>
  --
  * provider_id : CHAR(26) <<FK>>
  * region : VARCHAR(20)
  * base_url : VARCHAR(500)
  * weight : SMALLINT
}

entity CREDENTIAL_REF {
  * credential_ref_id : CHAR(26) <<PK>>
  --
  * provider_id : CHAR(26) <<FK>>
  * secret_ref : VARCHAR(200)
  * version : INT
  * state : VARCHAR(20)
  * rotated_at : TIMESTAMP
}

entity MODEL <<ES>> {
  * model_id : CHAR(26) <<PK>>
  --
  * provider_id : CHAR(26) <<FK>>
  * model_name : VARCHAR(100)
  * version : VARCHAR(40)
  * context_window : INT
  * max_output_tokens : INT
  * status : VARCHAR(20)
  * priority : SMALLINT
  regions : JSON
  <<UQ>> (provider_id, model_name, version)
}

entity CAPABILITY {
  * capability_id : VARCHAR(40) <<PK>>
  --
  * name : VARCHAR(100)
  * input_schema : JSON
  * output_schema : JSON
  * streamable : BOOLEAN
  * status : VARCHAR(20)
}

entity MODEL_CAPABILITY {
  * model_id : CHAR(26) <<PK,FK>>
  * capability_id : VARCHAR(40) <<PK,FK>>
  --
  constraints : JSON
}

entity MODEL_ALIAS <<ES>> {
  * alias_id : CHAR(26) <<PK>>
  --
  * name : VARCHAR(100)
  * tenant_id : CHAR(26)
  <<UQ>> (tenant_id, name)
}

entity ALIAS_TARGET {
  * alias_id : CHAR(26) <<PK,FK>>
  * model_id : CHAR(26) <<PK,FK>>
  --
  * weight : SMALLINT
}

entity PRICE_ENTRY {
  * price_entry_id : CHAR(26) <<PK>>
  --
  * model_id : CHAR(26) <<FK>>
  * input_per_1k : DECIMAL(12,6)
  * output_per_1k : DECIMAL(12,6)
  * currency : CHAR(3)
  * effective_from : TIMESTAMP
}

entity PLUGIN {
  * plugin_id : CHAR(26) <<PK>>
  --
  * name : VARCHAR(100)
  * version : VARCHAR(20)
  * spi_version : VARCHAR(20)
  * signature : VARCHAR(500)
  * status : VARCHAR(20)
}

entity SESSION {
  * session_id : CHAR(26) <<PK>>
  --
  * tenant_id : CHAR(26)
  * principal : VARCHAR(200)
  attributes : JSON
  * expires_at : TIMESTAMP
  * status : VARCHAR(20)
}

entity CONVERSATION {
  * conversation_id : CHAR(26) <<PK>>
  --
  * session_id : CHAR(26) <<FK>>
  * tenant_id : CHAR(26)
  title : VARCHAR(200)
  * status : VARCHAR(20)
  * turn_count : INT
}

entity TURN {
  * turn_id : CHAR(26) <<PK>>
  --
  * conversation_id : CHAR(26) <<FK>>
  * seq : INT
  * role : VARCHAR(10)
  * content_parts : JSON
  model_id : CHAR(26)
  usage : JSON
  * created_at : TIMESTAMP
  <<UQ>> (conversation_id, seq)
}

entity MEMORY {
  * memory_id : CHAR(26) <<PK>>
  --
  * tenant_id : CHAR(26)
  * scope : VARCHAR(20)
  * content : TEXT
  * embedding : VECTOR
  * importance : FLOAT
  ttl_at : TIMESTAMP
  * last_accessed_at : TIMESTAMP
}

entity ROUTING_POLICY <<ES>> {
  * policy_id : CHAR(26) <<PK>>
  --
  * scope : VARCHAR(20)
  tenant_id : CHAR(26)
  workflow_id : VARCHAR(100)
  * rules : JSON
  * version : INT
  * status : VARCHAR(20)
}

entity QUOTA_POLICY {
  * quota_id : CHAR(26) <<PK>>
  --
  * tenant_id : CHAR(26)
  * scope : VARCHAR(40)
  * period : VARCHAR(20)
  * limit_requests : BIGINT
  * limit_tokens : BIGINT
  * limit_cost : DECIMAL(14,4)
}

entity BUDGET {
  * budget_id : CHAR(26) <<PK>>
  --
  * tenant_id : CHAR(26)
  * period : VARCHAR(20)
  * limit_amount : DECIMAL(14,4)
  * currency : CHAR(3)
  * consumed : DECIMAL(14,4)
  * thresholds : JSON
}

entity USAGE_RECORD {
  * usage_id : CHAR(26) <<PK>>
  --
  * request_id : CHAR(26)
  * tenant_id : CHAR(26)
  * capability_id : VARCHAR(40)
  * provider_id : CHAR(26) <<FK>>
  * model_id : CHAR(26) <<FK>>
  * input_tokens : INT
  * output_tokens : INT
  * cost_amount : DECIMAL(12,6)
  * currency : CHAR(3)
  * duration_ms : INT
  * status : VARCHAR(20)
  * occurred_at : TIMESTAMP
}

entity AUDIT_RECORD {
  * audit_id : CHAR(26) <<PK>>
  --
  * request_id : CHAR(26)
  * trace_id : VARCHAR(64)
  * tenant_id : CHAR(26)
  * principal : VARCHAR(200)
  * capability_id : VARCHAR(40)
  model_alias : VARCHAR(100)
  * provider_id : CHAR(26)
  * model_id : CHAR(26)
  * routing_decision : JSON
  * request_digest : CHAR(64)
  response_digest : CHAR(64)
  request_body : TEXT
  * status : VARCHAR(20)
  error_code : VARCHAR(60)
  * usage : JSON
  * cost : JSON
  * duration_ms : INT
  * retries : SMALLINT
  * fallbacks : SMALLINT
  conversation_id : CHAR(26)
  * occurred_at : TIMESTAMP
}

entity BATCH_JOB <<ES>> {
  * job_id : CHAR(26) <<PK>>
  --
  * tenant_id : CHAR(26)
  * target_capability : VARCHAR(40)
  * status : VARCHAR(20)
  * total_items : INT
  * completed_items : INT
  * submitted_at : TIMESTAMP
}

entity BATCH_ITEM {
  * item_id : CHAR(26) <<PK>>
  --
  * job_id : CHAR(26) <<FK>>
  * seq : INT
  * request_payload : JSON
  result_payload : JSON
  * status : VARCHAR(20)
}

entity EVENT_STORE {
  * stream_id : CHAR(26) <<PK>>
  * version : BIGINT <<PK>>
  --
  * event_type : VARCHAR(60)
  * payload : JSON
  * occurred_at : TIMESTAMP
  * trace_id : VARCHAR(64)
}

PLUGIN ||--o{ PROVIDER
PROVIDER ||--|{ PROVIDER_ENDPOINT
PROVIDER ||--|{ CREDENTIAL_REF
PROVIDER ||--o{ MODEL
MODEL ||--|{ MODEL_CAPABILITY
CAPABILITY ||--o{ MODEL_CAPABILITY
MODEL_ALIAS ||--|{ ALIAS_TARGET
MODEL ||--o{ ALIAS_TARGET
MODEL ||--o{ PRICE_ENTRY
SESSION ||--o{ CONVERSATION
CONVERSATION ||--o{ TURN
BATCH_JOB ||--|{ BATCH_ITEM
PROVIDER ||--o{ USAGE_RECORD
MODEL ||--o{ USAGE_RECORD
@enduml
```

**設計注記**: AUDIT_RECORDは追記専用（UPDATE/DELETE権限を付与しない）。`request_body` は監査ポリシーopt-in時のみ格納（マスキング済）。USAGE_RECORDは集計Read Model（Cost/Analytics Query用）の元データで、時系列パーティション（月次）。EVENT_STOREは `(stream_id, version)` の楽観ロックで追記整合を担保。
